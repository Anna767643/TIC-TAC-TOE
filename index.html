type DBPool struct {
        pool *pgxpool.Pool
}

// NewDBPool - конструктор для di
func NewDBPool(dsn string) (*DBPool, error) {
        ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
        defer cancel()

        config, err := pgxpool.ParseConfig(dsn)
        if err != nil {
                return nil, fmt.Errorf("ошибка парсинга конфига: %w", err)
        }

        pool, err := pgxpool.NewWithConfig(ctx, config)
        if err != nil {
                return nil, fmt.Errorf("ошибка создания пула: %w", err)
        }
        if err := pool.Ping(ctx); err != nil {
                return nil, fmt.Errorf("не удалось пингануть БД: %w", err)
        }

        return &DBPool{pool: pool}, nil
}

// Close - закрывает соединение
func (db *DBPool) Close() {
        db.pool.Close()
}

// GetPool - передает соединение
func (db *DBPool) GetPool() *pgxpool.Pool {
        return db.pool
}

// insertIntoGames - добавляет новую игру в таблице games
// "repository: insert game %s: %w"

        sqlQuery := `
        INSERT INTO games(uuid, board, with_computer, state, next_player, winner, uuid_player_first, update_at)

        UPDATE games
        SET board = $1, state = $2, next_player = $3, winner = $4, uuid_player_second = $5, update_at = $6
        WHERE uuid = $7;
        `

        _, err := db.pool.Exec(ctx, sqlQuery, game.Board, game.State, game.NextPlayer, game.Winner, game.UUIDPlayerSecond, game.UpdatedAt, game.UUID)

                return fmt.Errorf("repository: update game %s: %w", game.UUID, err)

// selectAvaibleGamesFromGames - получение среза игр из таблицы games
// доступных для присоединения игроку с uuid = uuidUser
// "repository: select game for user %s: %w"
// "repository: scan game for user %s: %w"
func (db *DBPool) selectAvaibleGamesFromGames(ctx context.Context, uuidUser uuid.UUID) ([]gameEntity, error) {
        sqlQuery := `
        SELECT id, uuid, board, with_computer, state, next_player, winner, uuid_player_first, uuid_player_second, update_at from games
        WHERE uuid_player_first != $1 and state = 'wait users' and with_computer = false;
        `

        rows, err := db.pool.Query(ctx, sqlQuery, uuidUser)
        if err != nil {
                return []gameEntity{}, fmt.Errorf("repository: select game for user %s: %w", uuidUser, err)
        }
        defer rows.Close()
        games := make([]gameEntity, 0)

        for rows.Next() {
                var game gameEntity
                if err := rows.Scan(&game.ID, &game.UUID, &game.Board, &game.WithComputer, &game.State, &game.NextPlayer, &game.Winner, &game.UUIDPlayerFirst, &game.UUIDPlayerSecond, &game.UpdatedAt); err != nil {
                        return []gameEntity{}, fmt.Errorf("repository: scan game for user %s: %w", uuidUser, err)
                }
                games = append(games, game)
        }

        return games, nil
}

// selectFromGames - получение информации об игре по UUID игры из таблицы games
// ErrGameNotFound
// "repository: select game %s: %w"
func (db *DBPool) selectFromGames(ctx context.Context, uuid uuid.UUID) (gameEntity, error) {
        sqlQuery := `
        SELECT id, uuid, board, with_computer, state, next_player, winner, uuid_player_first, uuid_player_second, update_at from games
        WHERE uuid = $1;
        `

        row := db.pool.QueryRow(ctx, sqlQuery, uuid)
        var game gameEntity

        if err := row.Scan(&game.ID, &game.UUID, &game.Board, &game.WithComputer, &game.State, &game.NextPlayer, &game.Winner, &game.UUIDPlayerFirst, &game.UUIDPlayerSecond, &game.UpdatedAt); err != nil {
                if errors.Is(err, pgx.ErrNoRows) {
                        return gameEntity{}, domain.ErrGameNotFound
                }
                return gameEntity{}, fmt.Errorf("repository: select game %s: %w", uuid, err)
        }
        return game, nil
}

// toEntity - преобразует domain.Game в GameEntity
func toEntityGame(game domain.Game) gameEntity {
        return gameEntity{
                UUID:         game.UUID,
                Board:        toEntityBoard(game.Board),
                WithComputer: game.WithComputer,
                State:        game.State,
                NextPlayer:   game.NextPlayer,
                Winner:       game.Winner,

                UUIDPlayerFirst:  game.UUIDPlayerFirst,
                UUIDPlayerSecond: game.UUIDPlayerSecond,

                UpdatedAt: game.UpdatedAt,
        }
}

// toDomain - преобразует GameEntity в domain.Game
// если ID в хранилище не парситься в UUID, то возвращается ошибка
func toDomainGame(entity gameEntity) (domain.Game, error) {

        dboard, err := toDomainBoard(entity.Board)
        if err != nil {
                return domain.Game{}, err
        }

        return domain.Game{
                ID:           entity.ID,
                UUID:         entity.UUID,
                Board:        dboard,
                WithComputer: entity.WithComputer,
                State:        entity.State,
                NextPlayer:   entity.NextPlayer,
                Winner:       entity.Winner,

                UUIDPlayerFirst:  entity.UUIDPlayerFirst,
                UUIDPlayerSecond: entity.UUIDPlayerSecond,

                UpdatedAt: entity.UpdatedAt,
        }, nil
}

// toDomainBoard - преобразует строку в domain.Board
// если строка не парситься в int, то возвращается ошибка
func toDomainBoard(board string) (domain.Board, error) {
        parts := strings.Split(board, ", ")
        var dboard domain.Board
        n := 0

        for i := 0; i < 3; i++ {
                for j := 0; j < 3; j++ {
                        number, err := strconv.Atoi(parts[n])
                        if err != nil {
                                return domain.Board{}, err
                        }
                        dboard[i][j] = number
                        n++
                }
        }

        return dboard, nil
}

// toEntityBoard - преобразует domain.Board в строку
func toEntityBoard(board domain.Board) string {
        strArray := make([]string, 9)
        i := 0
        for _, r := range board {
                for _, value := range r {
                        strArray[i] = strconv.Itoa(value)
                        i++
                }
        }
        result := strings.Join(strArray, ", ")
        return result
}

SaveNewGame - сохранение новой игры
Возвращает новую игру
Ошибки:
ErrContextCanceled
"repository: insert game %s: %w"
"repository: failed select game after save: %w"
"repository: select game %s: %w"
"repository: failed to map game entity to domain: error"
*/
func (r *gameRepository) SaveNewGame(ctx context.Context, game domain.Game) (domain.Game, error) {
        if err := ctx.Err(); err != nil {
                return domain.Game{}, domain.ErrContextCanceled
        }

        gameEntity := toEntityGame(game)

        err := r.gs.insertIntoGames(ctx, gameEntity)
        if err != nil {
                return domain.Game{}, err
        }

        gameEntity, err = r.gs.selectFromGames(ctx, game.UUID)
        if err != nil {
                if err == domain.ErrGameNotFound {
                        return domain.Game{}, fmt.Errorf("repository: failed select game after save: %w", err)
                }
                return domain.Game{}, err
        }

        newGame, err := toDomainGame(gameEntity)
        if err != nil {
                return domain.Game{}, fmt.Errorf("repository: failed to map game entity to domain: %w", err)
        }

        return newGame, err
}

func newServer(cfg *Config, mux *http.ServeMux) *http.Server {
        return &http.Server{
                Addr:         cfg.Addr,
                Handler:      mux,
                ReadTimeout:  5 * time.Second,
                WriteTimeout: 10 * time.Second,
                IdleTimeout:  120 * time.Second,
        }
}

// runServer подключает сервер к жизненному циклу fx:
// OnStart - net.Listen + Serve в горутине, OnStop - Shutdown(ctx)
func runServer(lc fx.Lifecycle, srv *http.Server, db *datasource.DBPool) {
        lc.Append(fx.Hook{
                // OnStart выполняется при запуске приложения
                OnStart: func(ctx context.Context) error {
                        if err := datasource.CreateTableUsers(ctx, db.GetPool()); err != nil {
                                log.Fatalf("Failed to create users table: %v", err)
                        }
                        if err := datasource.CreateTableGames(ctx, db.GetPool()); err != nil {
                                log.Fatalf("Failed to create games table: %v", err)
                        }
                        log.Printf("Starting HTTP server on %s", srv.Addr)
                        // Запускаем сервер в отдельной горутине, чтобы не блокировать старт fx
                        go func() {
                                if err := srv.ListenAndServe(); err != nil && err != http.ErrServerClosed {
                                        log.Printf("HTTP server ListenAndServe error: %v", err)
                                }
                        }()
                        return nil
                },

                // OnStop выполняется при завершении приложения (например, по SIGTERM)
                OnStop: func(ctx context.Context) error {
                        log.Println("Shutting down HTTP server...")
                        // Graceful shutdown: дает 5 секунд на завершение текущих запросов
                        shutdownCtx, cancel := context.WithTimeout(ctx, 5*time.Second)
                        defer cancel()
                        err := srv.Shutdown(shutdownCtx)

                        db.Close()
                        return err
                },
        })
}

// IsFinished сообщает, окончена ли игра, и кто победил:
// player = X, O или Empty при ничьей / незавершённой игре.
func (s *gameService) isFinished(board Board) (bool, int) {
        var finished bool
        var player int

        if finished = s.isWin(board, X); finished {
                player = X
        } else if finished = s.isWin(board, O); finished {
                player = O
        } else if finished = s.isFilledBoard(board); finished {
                player = Empty
        }

        return finished, player
}

// isWin проверяет выйграл ли определенный игрок
func (s *gameService) isWin(board Board, winner int) bool {
        var (
                count int
                win   bool
        )

        //Проходим по горизонталям
        for idx := range board {
                for _, val := range board[idx] {
                        if val == winner {
                                count++
                        }
                        if count == Size {
                                win = true
                                return win
                        }
                }
                count = 0
        }

        //Проходим по вертикалям
        for i := 0; i < Size; i++ {
                for idx := range board {
                        if board[idx][i] == winner {
                                count++
                        }
                        if count == Size {
                                win = true
                                return win
                        }
                }
                count = 0
        }

        //Крест-накрест
        if (board[0][0] == winner && board[1][1] == winner && board[2][2] == winner) ||
                (board[0][2] == winner && board[1][1] == winner && board[2][0] == winner) {
                win = true
                return win
        }

        return win
}

// isFilledBoard проверяет заполнено ли поле полностью (в случаях ничьи бывает)
func (s *gameService) isFilledBoard(board Board) bool {
        var (
                count int
                over  bool
        )

        for idx := range board {
                for _, val := range board[idx] {
                        if val == 0 {
                                count++
                        }
                }
        }

        if count == 0 {
                over = true
        }

        return over
}

        j int

/*
NextMove - выполняет ход игроком
Если следующий ход компьютера, то он выполняется с помощью алгоритма минимакс
Возвращает игру с обновлённым полем.
Ошибки:
ErrContextCanceled - игра прерывается
ErrGameFinished - игра была ранее окончена
ErrWaitOpponent - второй игрок еще не присоеденился к игре
ErrGameNotFound - игра не найдена
ErrDoesntHaveSuchUser - в этой игре нет такого пользователя
ErrOpponentsMove - ход аппонента
ErrInvalidBoard - поле нарушает правила или изменены прошлые ходы
ErrInvalidCell - недопустимое значение клетки
*/
func (s *gameService) NextMove(ctx context.Context, board Board, uuidGame uuid.UUID, uuidUser uuid.UUID) (Game, error) {
        if err := ctx.Err(); err != nil {
                return Game{}, ErrContextCanceled
        }

        // Сперва получим инфу об игре вместе с игроком
        game, err := s.repo.Get(ctx, uuidGame)
        if err != nil {
                return Game{}, err
        }

        // Проверяем состояние игры
        switch game.State {
        case StateWaitUsers:
                return Game{}, ErrWaitOpponent
        case StateDraw:
                return Game{}, ErrGameFinished
        case StateWin:
                return Game{}, ErrGameFinished
        }

        // Сверяем UUID игрока
        isFirst := game.UUIDPlayerFirst == uuidUser
        isSecond := game.UUIDPlayerSecond != nil && *game.UUIDPlayerSecond == uuidUser
        // Данный игрок в эту игру не играет
        if !isFirst && !isSecond {
                return Game{}, ErrDoesntHaveSuchUser
        }
        if game.NextPlayer != nil && *game.NextPlayer != uuidUser {
                return Game{}, ErrOpponentsMove
        }

        var playerCellValue int
        var playerNextMove uuid.UUID

        // Устанавливаем чем он должен ходить
        if game.UUIDPlayerFirst == uuidUser {
                playerCellValue = X
                if game.UUIDPlayerSecond != nil {
                        playerNextMove = *game.UUIDPlayerSecond
                }
        } else {
                playerCellValue = O
                playerNextMove = game.UUIDPlayerFirst
        }

        // Проверяем что ход допустимый
        if err := s.validate(board, game.Board, playerCellValue); err != nil {
                return Game{}, err
        }

        game.Board = board

        // Проверяем не окончена ли игра после этого хода
        if finished, winner := s.isFinished(game.Board); finished {
                switch winner {
                case Empty:
                        game.State = StateDraw
                default:
                        game.State = StateWin
                        game.Winner = &uuidUser
                }
                game.UpdatedAt = time.Now()
                if err := s.repo.Save(ctx, game); err != nil {
                        return Game{}, err
                }
                return game, nil
        }

        // Если игра с компьютером
        if game.WithComputer {
                return s.nextMoveWithComputer(ctx, game)
        }

        // Записываем, что следующий ход за оппонентом
        game.State = StateNextMove
        game.NextPlayer = &playerNextMove
        game.UpdatedAt = time.Now()
        if err := s.repo.Save(ctx, game); err != nil {
                return Game{}, err
        }

        return game, nil
}

/*
nextMoveWithComputer -совершает следующий ход компьютером и записывает результат
Возращает новую игру
Ошибки:
- ErrContextCanceled
- "repository: update game %s: %w"
*/
func (s *gameService) nextMoveWithComputer(ctx context.Context, game Game) (Game, error) {
        bestScore := -1000
        var bestPosition pos

        // Перебираем все возможные ходы для O (компьютер)
        for _, position := range s.searchAllOptions(game.Board) {
                newBoard := game.Board
                newBoard[position.i][position.j] = O

                //Уходим в рекурсию для подсчета выгодности данного хода
                score, err := s.minimax(ctx, newBoard, 0, false)
                if err != nil {
                        return Game{}, err
                }

                // Если счет лучший, то сохраняем
                if score > bestScore {
                        bestScore = score
                        bestPosition = position
                }
        }

        // Устанавливаем позицию в копию и возвращаем
        newGame := game
        newGame.Board[bestPosition.i][bestPosition.j] = O
        newGame.UpdatedAt = time.Now()

        if finished, winner := s.isFinished(newGame.Board); finished {
                switch winner {
                case Empty:
                        newGame.State = StateDraw
                default:
                        newGame.State = StateWin
                }
        }

        if err := s.repo.Save(ctx, newGame); err != nil {
                return Game{}, err
        }

        return newGame, nil
}

// minimax возвращает оценку позиции для O: чем выше, тем лучше.
// Флаг isMax показывает чей ход, компьютера или игрока
func (s *gameService) minimax(ctx context.Context, b Board, depth int, isMax bool) (int, error) {
        if err := ctx.Err(); err != nil {
                return 0, ErrContextCanceled
        }

        // В зависимости от выиграшевого игрока уcтанавливаем счёт с поправкой на глубину
        if finished, winner := s.isFinished(b); finished {
                switch winner {
                case O:
                        return 10 - depth, nil
                case X:
                        return depth - 10, nil
                default:
                        return 0, nil
                }
        }

        //Уставливаем счет и игрока в зависимости от isMax
        bestScore := -1000
        player := O
        if !isMax {
                bestScore = 1000
                player = X
        }

        // Перебираем все позиции и ищем самую лучшую
        for _, position := range s.searchAllOptions(b) {
                newBoard := b
                newBoard[position.i][position.j] = player

                score, err := s.minimax(ctx, newBoard, depth+1, !isMax)
                if err != nil {
                        return 0, ErrContextCanceled
                }

                if isMax && score > bestScore {
                        bestScore = score
                } else if !isMax && score < bestScore {
                        bestScore = score
                }
        }

        // Возвращаем счет для определенной позиции
        return bestScore, nil
}

// searchAllOptions - ищет все доступные пустые клетки на доске.
func (s *gameService) searchAllOptions(board Board) []pos {
        options := make([]pos, 0, 9)
        for i := range board {
                for j := range board[i] {
                        if board[i][j] == 0 {
                                options = append(options, pos{i, j})
                        }
                }
        }
        return options
}

/*
Register - регистрирует нового пользователя
возвращает факт успешной регистрации
Ошибки:
ErrContextCanceled
"domain: faled to generate password hash %w"
ErrLoginIncorrect
"repository: insert user with %s: %w"
"repository: select user with %s: %w"
"domain: faled to get user %w"
*/
func (u *userService) Register(ctx context.Context, signUpRequest SignUpRequest) (User, error) {
        if err := ctx.Err(); err != nil {
                return User{}, ErrContextCanceled
        }

        passwordHash, err := bcrypt.GenerateFromPassword([]byte(signUpRequest.Password), 10)
        if err != nil {
                return User{}, fmt.Errorf("domain: faled to generate password hash %w", err)
        }

        user := User{
                UUID:         uuid.New(),
                Login:        signUpRequest.Login,
                PasswordHash: string(passwordHash),
                CreateAt:     time.Now(),
        }

        if err := u.repo.Save(ctx, user); err != nil {
                return User{}, err
        }

        user, err = u.repo.Get(ctx, signUpRequest.Login, uuid.UUID{})
        if err != nil {
                if err == ErrUserNotFound {
                        return User{}, fmt.Errorf("domain: faled to get user %w", err)

                }
                return User{}, err
        }

        user.PasswordHash = ""

        return user, nil
}

/*
LogIn - авторизирует пользователя
возвращает UUIDпользователя
Ошибки:
ErrContextCanceled
ErrLoginOrPasswordIncorrect
"repository: select user with %s: %w"
*/
func (u *userService) LogIn(ctx context.Context, signUpRequest SignUpRequest) (uuid.UUID, error) {
        if err := ctx.Err(); err != nil {
                if err == ErrUserNotFound {
                        return uuid.UUID{}, ErrContextCanceled
                }
                return uuid.UUID{}, err
        }

        user, err := u.repo.Get(ctx, signUpRequest.Login, uuid.UUID{})
        if err != nil {
                return uuid.UUID{}, ErrLoginOrPasswordIncorrect
        }

        if err := bcrypt.CompareHashAndPassword([]byte(user.PasswordHash), []byte(signUpRequest.Password)); err != nil {
                return uuid.UUID{}, ErrLoginOrPasswordIncorrect
        }

        return user.UUID, nil
}

// Validate - проверяет не изменены ли предыдущие ходы и новое значение равно только playerCellValue
// ErrInvalidCell - значения вне диапозона
// ErrInvalidBoard - значение не playerCellValue, больше одного изменения или поле не совпадает
func (s *gameService) validate(incoming, stored Board, playerCellValue int) error {
        var count int

        for i := range stored {
                for j := range stored[i] {
                        if incoming[i][j] > O || incoming[i][j] < Empty {
                                return ErrInvalidCell
                        }
                        if stored[i][j] == Empty && incoming[i][j] != Empty {
                                if incoming[i][j] != playerCellValue {
                                        return ErrInvalidBoard
                                }
                                count++
                        }
                        if stored[i][j] != Empty && stored[i][j] != incoming[i][j] {
                                return ErrInvalidBoard
                        }
                }
        }

        if count != 1 {
                return ErrInvalidBoard
        }

        return nil
}

*/
func (h *GameHandler) handlePatchCurrentGame(w http.ResponseWriter, r *http.Request) {
        ctx, uuidUser, err := getUUIDUser(w, r)
        if err != nil {
                return
        }

        gameId := r.PathValue("uuid")

        uuidGame, err := toUUID(gameId)
        if err != nil {
                writeError(w, http.StatusBadRequest, err.Error())
                return
        }

        var rboard reqBoard

        if err := json.NewDecoder(r.Body).Decode(&rboard); err != nil {
                writeError(w, http.StatusBadRequest, "wrong body "+err.Error())
                return
        }

        game, err := h.gameService.NextMove(ctx, domain.Board(rboard.BoardDTO), uuidGame, uuidUser)
        if err == domain.ErrGameNotFound {

                writeError(w, http.StatusNotFound, err.Error())
                return
        } else if err == domain.ErrInvalidBoard || err == domain.ErrInvalidCell || err == domain.ErrDoesntHaveSuchUser {

                writeError(w, http.StatusBadRequest, "wrong body "+err.Error())
                return
        } else if err == domain.ErrOpponentsMove || err == domain.ErrGameFinished {

                writeError(w, http.StatusConflict, err.Error())
                return
        } else if err != nil {

                writeError(w, http.StatusInternalServerError, err.Error())
                return
        }

        switch game.State {
        case domain.StateDraw:
                writeJSON(w, http.StatusOK, map[string]string{
                        "message": "GAME OVER",
                        "winner":  "DRAW",
                })
        case domain.StateWin:
                writeJSON(w, http.StatusOK, map[string]string{
                        "message": "GAME OVER",
                        "winner":  game.Winner.String(),
                })
        default:
                gameDTO := toGameDTO(game)
                writeJSON(w, http.StatusOK, gameDTO)

        }
}

- 400 Bad Request
- 404 Not Found
- 500 Internal Server Error
JSON, описание ошибки
*/
func (h *GameHandler) handleGetGameInfo(w http.ResponseWriter, r *http.Request) {

        if err != nil {
                return
        }

        gameId := r.PathValue("uuid")

        uuidGame, err := toUUID(gameId)
        if err != nil {
                writeError(w, http.StatusBadRequest, err.Error())

        }

        game, err := h.gameService.GetGame(ctx, uuidGame, uuidUser)

        if err == domain.ErrGameNotFound {

                return
        } else if err == domain.ErrDoesntHaveSuchUser {

/*
HandleCreateUser - Хэндел для регистрация нового пользователя

Method POST /users
JSON в теле запроса: login и passwort

Ответ:
Успешно
Status code:
- 201 Created
JSON, представляющий созданного пользователя, в теле ответа

Ошибка
Status code:
- 400 Bad Request
- 409 Conflict
- 500 Internal Server Error
JSON, описание ошибки
*/
func (u *UserHandler) handleUsersCreate(w http.ResponseWriter, r *http.Request) {
        ctx := r.Context()

        var signUpRequest domain.SignUpRequest
        if err := json.NewDecoder(r.Body).Decode(&signUpRequest); err != nil {
                writeError(w, http.StatusBadRequest, "wrong body, need login and password in json, error: "+err.Error())
                return
        }

        if signUpRequest.Login == "" || signUpRequest.Password == "" {
                writeError(w, http.StatusBadRequest, "wrong body, login or password not be empty")
                return
        }

        // В зависимости от ошибки - разные ответы, либо Conflict, либо Internal Server Error
        user, err := u.userService.Register(ctx, signUpRequest)
        if err == domain.ErrLoginIncorrect {
                writeError(w, http.StatusConflict, err.Error())
                return
        } else if err != nil {
                writeError(w, http.StatusInternalServerError, err.Error())
                return
        }

        userDTO := toUserDTO(user)
        writeJSON(w, http.StatusCreated, userDTO)
}

// toUUID - парсит UUID
// Возвращает UUID или ErrInvalidUUID
func toUUID(str string) (uuid.UUID, error) {
        uuid, err := uuid.Parse(str)
        if err != nil {
                return uuid, domain.ErrInvalidUUID
        }

        return uuid, nil
}

// toDTO - преобразует domain.Game в GameDTO
func toGameDTO(game domain.Game) gameDTO {
        return gameDTO{
                ID:           game.ID,
                UUID:         game.UUID.String(),
                Board:        boardDTO(game.Board),
                WithComputer: game.WithComputer,
                State:        game.State,
                NextPlayer:   uuidPtrToString(game.NextPlayer),
                Winner:       uuidPtrToString(game.Winner),

                UUIDPlayerFirst:  game.UUIDPlayerFirst.String(),
                UUIDPlayerSecond: uuidPtrToString(game.UUIDPlayerSecond),
                UpdatedAt:        game.UpdatedAt.UTC().Format(time.RFC3339),
        }
}

// uuidPtrToString - переводит *uuid.UUID в *string
func uuidPtrToString(val *uuid.UUID) *string {
        if val == nil {
                return nil
        }
        str := val.String()
        return &str
}

                CreateAt: user.CreateAt.UTC().Format(time.RFC3339),
        }
}

// contextKey - тип для ключей контекста
type contextKey string

// Константа-ключ
const (
        userUUIDKey contextKey = "userUUID"
)

// UserAuthenticator - защита от неавторизованных запросов
type UserAuthenticator struct {
        userService domain.UserService
}

// NewUserAuthenticator - конструктор для di
func NewUserAuthenticator(userService domain.UserService) *UserAuthenticator {
        return &UserAuthenticator{userService: userService}
}

// Validate - это Middleware. Он принимает следующий хендлер и возвращает новый
func (auth *UserAuthenticator) Validate(next http.HandlerFunc) http.HandlerFunc {
        return func(w http.ResponseWriter, r *http.Request) {
                ctx := r.Context()

                login, password, ok := r.BasicAuth()
                if !ok {
                        writeError(w, http.StatusBadRequest, "Need authorization")
                        return
                }

                signUpRequest := domain.SignUpRequest{
                        Login:    login,
                        Password: password,
                }

                uuid, err := auth.userService.LogIn(ctx, signUpRequest)
                if err == domain.ErrLoginOrPasswordIncorrect {
                        writeError(w, http.StatusUnauthorized, err.Error())
                        return
                } else if err != nil {
                        writeError(w, http.StatusInternalServerError, err.Error())
                        return
                }

                // Складываем наше значение в контекст
                newCtx := context.WithValue(ctx, userUUIDKey, uuid)

                next(w, r.WithContext(newCtx))
        }
}
